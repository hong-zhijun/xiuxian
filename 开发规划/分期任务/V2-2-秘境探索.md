# V2-2：秘境探索（简化战斗系统）

**前置：V2-1 宗门升级已完成。**

3~5 人自用娱乐游戏。本任务加入秘境探索——派弟子进秘境，自动计算胜负，成功拿奖励，失败弟子受伤。

**核心原则：简单直接。没有回合制，没有技能，没有装备。弟子属性 vs 秘境难度 → 一个概率 → 出结果。**

## 一、数据库迁移 (migrations/0006_explorations.sql)

```sql
CREATE TABLE explorations (
  id TEXT PRIMARY KEY,
  sect_id TEXT NOT NULL REFERENCES sects(id),
  realm_id TEXT NOT NULL,
  party TEXT NOT NULL DEFAULT '[]',
  success INTEGER NOT NULL DEFAULT 0,
  rewards TEXT NOT NULL DEFAULT '{}',
  created_at INTEGER NOT NULL
);

CREATE INDEX explorations_sect_idx ON explorations(sect_id, created_at DESC);
```

字段说明：
- `realm_id`：秘境 id（对应下面的秘境定义，不是弟子境界）
- `party`：JSON 数组 `["discipleId1", "discipleId2"]`
- `success`：0=失败，1=成功
- `rewards`：JSON `{"spiritStone": "50000", "herb": "20000"}`（最小单位，成功时的实际奖励）

在 `apps/server/src/infra/db/readiness.ts` 的 REQUIRED_TABLES 里加上 `explorations`。

## 三、战力计算 (在 realms.ts 里)

弟子战力由**境界阶段**和**资质**决定：

```typescript
import { REALMS, realmIndex } from './constants';

/**
 * 单个弟子战力 = 境界基础值 × (1 + 资质/100)。
 * 境界基础值 = (境界序号×3 + 阶段) × 10。
 * 例：炼气一层 = (0×3+1)×10 = 10，金丹三层 = (2×3+3)×10 = 90。
 */
export function discipleCombatPower(realmId: string, stage: number, aptitude: number): number {
  const rIdx = realmIndex(realmId);
  const base = (rIdx * 3 + stage) * 10;
  return Math.floor(base * (1 + aptitude / 100));
}

/**
 * 队伍总战力。
 */
export function partyCombatPower(
  members: { realmId: string; stage: number; aptitude: number }[],
): number {
  return members.reduce((sum, m) => sum + discipleCombatPower(m.realmId, m.stage, m.aptitude), 0);
}

/**
 * 演武场加成：每级 +10% 战力。
 */
export const ARENA_COMBAT_BONUS_BP_PER_LEVEL = 1000;

/**
 * 成功率（基点 0~9500）= power / (power + difficulty) × 10000，clamp 到 [500, 9500]。
 * 演武场等级加成后再 clamp。
 */
export function explorationSuccessChanceBp(
  power: number,
  difficulty: number,
  arenaLevel: number = 0,
): number {
  const bonusBp = arenaLevel * ARENA_COMBAT_BONUS_BP_PER_LEVEL;
  const effectivePower = Math.floor(power * (10_000 + bonusBp) / 10_000);
  const raw = Math.floor((effectivePower / (effectivePower + difficulty)) * 10_000);
  return Math.min(9500, Math.max(500, raw));
}
```

```typescript
export interface ExplorationRow {
  id: string;
  sect_id: string;
  realm_id: string;
  party: string;
  success: number;
  rewards: string;
  created_at: number;
}

export class ExplorationRepository extends ParamRepository {
  async countTodayBySectAndRealm(sectId: string, realmId: string, dayStartMs: number): Promise<number> {
    const row = await this.one<{ total: number }>({
      sql: 'SELECT COUNT(*) AS total FROM explorations WHERE sect_id = ? AND realm_id = ? AND created_at >= ?',
      params: [sectId, realmId, dayStartMs],
    });
    return Number(row?.total ?? 0);
  }

  async findRecentBySectId(sectId: string, limit: number): Promise<ExplorationRow[]> {
    return this.all<ExplorationRow>({
      sql: `SELECT id, sect_id, realm_id, party, success, rewards, created_at
            FROM explorations WHERE sect_id = ? ORDER BY created_at DESC LIMIT ?`,
      params: [sectId, limit],
    });
  }
}

export function insertExplorationStatement(row: {
  id: string;
  sectId: string;
  realmId: string;
  party: string;
  success: boolean;
  rewards: string;
  now: number;
}): ParameterizedQuery {
  return {
    sql: `INSERT INTO explorations (id, sect_id, realm_id, party, success, rewards, created_at)
          VALUES (?, ?, ?, ?, ?, ?, ?)`,
    params: [row.id, row.sectId, row.realmId, row.party, row.success ? 1 : 0, row.rewards, row.now],
  };
}
```

## 五、探索逻辑 (修改 apps/server/src/modules/game/service.ts)

新增 `exploreSectRealm` 函数：

```typescript
export async function exploreSectRealm(
  db: D1Database,
  userId: string,
  realmId: string,
  discipleIds: string[],
  now: number,
): Promise<{ state: SectStateView; result: ExplorationResultView }> {
  const draft = await draftFor(db, userId, now);
  const realm = findSecretRealm(realmId);
  if (!realm) throw new AppError('NOT_FOUND', '秘境不存在');

  // 1. 宗门等级检查
  if (draft.sect.level < realm.requiredSectLevel) {
    throw new AppError('INVALID_STATUS', `需要宗门 ${realm.requiredSectLevel} 级才能进入`);
  }

  // 2. 演武场检查：宗门必须有演武场才能探索（V2-1 在 4 级解锁）
  const arena = draft.buildings.find(b => b.def_id === 'arenaHall');
  if (!arena) {
    throw new AppError('INVALID_STATUS', '需要先建造演武场');
  }

  // 3. 队伍人数检查
  if (discipleIds.length < realm.minParty || discipleIds.length > realm.maxParty) {
    throw new AppError('VALIDATION_ERROR', `需要 ${realm.minParty}~${realm.maxParty} 名弟子`);
  }

  // 4. 去重检查
  if (new Set(discipleIds).size !== discipleIds.length) {
    throw new AppError('VALIDATION_ERROR', '不能派遣重复弟子');
  }

  // 5. 弟子存在性与状态检查（不能是受伤弟子）
  const members: { realmId: string; stage: number; aptitude: number; name: string }[] = [];
  for (const id of discipleIds) {
    const disciple = draft.discipleById(id);
    if (disciple.injured_until !== null && Number(disciple.injured_until) > now) {
      throw new AppError('INVALID_STATUS', `${disciple.name}正在疗伤，无法出战`);
    }
    members.push({
      realmId: disciple.realm_id,
      stage: disciple.stage,
      aptitude: Number(disciple.aptitude),
      name: disciple.name,
    });
  }

  // 6. 每日次数检查
  if (realm.dailyLimit !== null) {
    const dayStart = dayStartMs(now);
    const used = await new ExplorationRepository(db).countTodayBySectAndRealm(
      draft.sect.id, realmId, dayStart,
    );
    if (used >= realm.dailyLimit) {
      throw new AppError('DAILY_LIMIT', `今日${realm.name}探索次数已用完（${realm.dailyLimit}次/天）`);
    }
  }

  // 7. 扣资源（入场费）
  for (const [resourceId, amount] of Object.entries(realm.entryCost)) {
    draft.requireResource(resourceId, Number(amount));
  }

  // 8. 计算战力与成功率
  const power = partyCombatPower(members);
  const arenaLevel = arena.level;
  const chanceBp = explorationSuccessChanceBp(power, realm.difficulty, arenaLevel);
  const roll = Math.floor(Math.random() * 10_000);
  const success = roll < chanceBp;

  // 9. 成功：发放奖励
  const actualRewards: Record<string, string> = {};
  if (success) {
    for (const [resourceId, amount] of Object.entries(realm.rewards)) {
      draft.addStatement(resourceDeltaStatement(draft.sect.id, resourceId, Number(amount), now));
      // 内存里也要加
      draft.balances = draft.balances.map(row =>
        row.resource_id === resourceId
          ? { ...row, balance: Number(row.balance) + Number(amount), updated_at: now }
          : row,
      );
      actualRewards[resourceId] = amount;
    }
  }

  // 10. 失败：弟子受伤（冷却 10 分钟）
  const INJURY_DURATION_MS = 10 * 60 * 1000;
  if (!success) {
    for (const id of discipleIds) {
      const injuredUntil = now + INJURY_DURATION_MS;
      draft.addStatement(updateDiscipleInjuryStatement(id, injuredUntil));
      const d = draft.disciples.find(dd => dd.id === id);
      if (d) d.injured_until = injuredUntil;
    }
  }

  // 11. 写入探索记录
  const explorationId = crypto.randomUUID();
  draft.addStatement(insertExplorationStatement({
    id: explorationId,
    sectId: draft.sect.id,
    realmId,
    party: JSON.stringify(discipleIds),
    success,
    rewards: JSON.stringify(actualRewards),
    now,
  }));

  await draft.commit();
  return {
    state: draft.view(),
    result: {
      realmName: realm.name,
      success,
      chanceBp,
      roll,
      rewards: actualRewards,
      memberNames: members.map(m => m.name),
      message: success
        ? `探索${realm.name}成功！获得丰厚奖励。`
        : `探索${realm.name}失败，弟子受伤需疗养。`,
    },
  };
}
```

```typescript
/** UTC+8 当天 0 点的毫秒时间戳，用于计算每日限次。 */
function dayStartMs(now: number): number {
  const utc8Ms = now + 8 * 3_600_000;
  const dayMs = 86_400_000;
  return Math.floor(utc8Ms / dayMs) * dayMs - 8 * 3_600_000;
}
```

```typescript
  };
}
```

```typescript
export async function listSecretRealms(
  db: D1Database,
  userId: string,
  now: number,
): Promise<SecretRealmListView[]> {
  const snapshot = await loadSnapshot(db, userId);
  if (!snapshot) return [];

  const sectLevel = snapshot.sect.level;
  const dayStart = dayStartMs(now);
  const arena = snapshot.buildings.find(b => b.def_id === 'arenaHall');
  const arenaLevel = arena?.level ?? 0;

  const views: SecretRealmListView[] = [];
  for (const realm of SECRET_REALMS) {
    const locked = sectLevel < realm.requiredSectLevel;
    let usedToday = 0;
    if (realm.dailyLimit !== null && !locked) {
      usedToday = await new ExplorationRepository(db).countTodayBySectAndRealm(
        snapshot.sect.id, realm.id, dayStart,
      );
    }
    views.push({
      id: realm.id,
      name: realm.name,
      description: realm.description,
      difficulty: realm.difficulty,
      entryCost: realm.entryCost,
      rewards: realm.rewards,
      minParty: realm.minParty,
      maxParty: realm.maxParty,
      dailyLimit: realm.dailyLimit,
      usedToday,
      requiredSectLevel: realm.requiredSectLevel,
      locked,
      hasArena: !!arena,
    });
  }
  return views;
}
```

### 10.2 ExplorePanel 逻辑

```typescript
const props = defineProps<{
  state: SectStateView;
  busy: boolean;
}>();

const emit = defineEmits<{
  explore: [realmId: string, discipleIds: string[]];
}>();

const realms = ref<SecretRealmView[]>([]);
const selectedParty = ref<Record<string, string[]>>({});

// 可用弟子（未受伤）
const availableDisciples = computed(() =>
  props.state.disciples.filter(d => d.injuredUntil === null || new Date(d.injuredUntil) <= new Date()),
);

// 资源名映射（从 state 里取）
const resourceNameMap = computed(() =>
  Object.fromEntries(props.state.resources.map(r => [r.id, r.name])),
);

// 加载秘境列表
async function loadRealms() {
  realms.value = await fetchSecretRealms();
}

// 初始加载 + state 变化时刷新
watch(() => props.state, loadRealms, { immediate: true });

function canExplore(realm: SecretRealmView): boolean {
  if (props.busy || realm.locked || !realm.hasArena) return false;
  const party = selectedParty.value[realm.id] ?? [];
  if (party.length < realm.minParty || party.length > realm.maxParty) return false;
  if (realm.dailyLimit !== null && realm.usedToday >= realm.dailyLimit) return false;
  return true;
}

function doExplore(realmId: string): void {
  const party = selectedParty.value[realmId] ?? [];
  if (party.length === 0) return;
  emit('explore', realmId, party);
}

function realmGlyph(id: string): string {
  const map: Record<string, string> = {
    mistyForest: '雾', savageMine: '矿', fallenStarAbyss: '渊',
    beastNest: '兽', ancientRealm: '古', tribulationRuins: '劫',
  };
  return map[id] ?? '境';
}
```

```html
<ExplorePanel
  :state="state"
  :busy="busy"
  @explore="handleExplore"
/>
```

### 10.4 App.vue 处理探索事件

```typescript
async function handleExplore(realmId: string, discipleIds: string[]) {
  busy.value = true;
  try {
    const { state: newState, result } = await explore(realmId, discipleIds);
    state.value = newState;
    const tone = result.success ? 'success' : 'warning';
    emit('notify', tone, result.realmName, result.message);
  } catch (e: any) {
    emit('notify', 'warning', '探索失败', e.message ?? '无法探索');
  } finally {
    busy.value = false;
  }
}
```

2. `npm run build` 通过
