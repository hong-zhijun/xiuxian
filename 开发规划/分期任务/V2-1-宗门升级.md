# V2-1：宗门升级系统 + 境界扩展 + 新建筑

3~5 人自用娱乐游戏。已有基座：完整挂机循环（资源/弟子/建筑/突破）+ 随机事件（P3）。

本任务给现有挂机循环加**长期目标**：宗门升级（条件门控，非纯花资源）、新建筑（藏经阁加速修炼）、更长的境界链。

**核心原则：简单直接。宗门等级就是一个条件判定 + 数值提升，没有科技树、没有前置任务链。**

```typescript
export function findSectLevel(level: number): SectLevelDef {
  return SECT_LEVELS[level - 1] ?? SECT_LEVELS[0]!;
}

export function nextSectLevel(currentLevel: number): SectLevelDef | null {
  return currentLevel >= MAX_SECT_LEVEL ? null : (SECT_LEVELS[currentLevel] ?? null);
}

/** 境界在 REALMS 数组中的下标，用于比较高低。 */
export function realmIndex(realmId: string): number {
  const idx = REALMS.findIndex((r) => r.id === realmId);
  return idx < 0 ? 0 : idx;
}

/** 计算实际资源容量 = 基础容量 × 倍率（取整）。 */
export function effectiveCapacity(baseCapacity: string, multiplier: number): number {
```

```typescript
export function cultivationRatePerHour(
  config: GameConfigContent,
  disciple: DiscipleState,
  scriptureLibraryLevel: number = 0,
): number {
  if (disciple.assignment !== 'cultivating') return 0;
  const rawBonusBp = disciple.aptitude * config.cultivation.aptitudeCoefficientPerPointBp;
  const bonusBp = Math.min(rawBonusBp, config.cultivation.maxTotalBonusBp);
  const coefficientBp = config.cultivation.aptitudeCoefficientBaseBp + bonusBp;
  const baseRate = Math.floor((config.cultivation.baseRatePerHour * coefficientBp) / 10_000);
  // 藏经阁加成：每级 +10%
  const libraryBonusBp = scriptureLibraryLevel * SCRIPTURE_LIBRARY_CULTIVATION_BONUS_BP_PER_LEVEL;
  return Math.floor(baseRate * (10_000 + libraryBonusBp) / 10_000);
}
```

### 5.3 新增 upgradeSect 函数

```typescript
export async function upgradeSect(
  db: D1Database,
  userId: string,
  now: number,
): Promise<SectStateView> {
  const draft = await draftFor(db, userId, now);
  const currentLevel = Number(draft.sect.level);
  const next = nextSectLevel(currentLevel);
  if (next === null) {
    throw new AppError('INVALID_STATUS', '宗门已达最高品阶');
  }

  // 1. 检查资源
  for (const [resourceId, amount] of Object.entries(next.upgradeCost)) {
    draft.requireResource(resourceId, Number(amount));
  }

  // 2. 检查建筑条件
  for (const req of next.buildingRequirements) {
    const building = draft.buildings.find(b => b.def_id === req.defId);
    const buildingName = draft.config.buildings.find(b => b.id === req.defId)?.name ?? req.defId;
    if (!building || building.level < req.minLevel) {
      throw new AppError('INVALID_STATUS', `${buildingName}需要 ${req.minLevel} 级（当前 ${building?.level ?? 0} 级）`);
    }
  }

  // 3. 检查弟子条件
  for (const req of next.discipleRequirements) {
    const realmReqIdx = realmIndex(req.minRealmId);
    const qualified = draft.disciples.filter(d => realmIndex(d.realm_id) >= realmReqIdx).length;
    const realmName = findRealm(req.minRealmId).name;
    if (qualified < req.count) {
      throw new AppError('INVALID_STATUS', `需要 ${req.count} 名${realmName}及以上弟子（当前 ${qualified} 名）`);
    }
  }

  // 4. 升级宗门
  draft.sect.level = next.level;
  draft.addStatement(updateSectLevelStatement(draft.sect.id, next.level));

  // 5. 自动解锁新建筑
  for (const defId of next.unlockBuildings) {
    if (draft.buildings.find(b => b.def_id === defId)) continue; // 已存在则跳过
    const buildingRow: BuildingRow = {
      id: crypto.randomUUID(),
      sect_id: draft.sect.id,
      def_id: defId,
      level: 1,
      created_at: now,
    };
    draft.buildings.push(buildingRow);
    draft.addStatement(insertBuildingStatement({
      id: buildingRow.id, sectId: draft.sect.id, defId, level: 1, now,
    }));
  }

  // 6. 更新 capacity（升级后立即生效）
  const newLevelDef = findSectLevel(next.level);
  // draft 里的 discipleCapacity/buildingCapacity 在 view() 里会重新从等级查

  await draft.commit();
  return draft.view();
}
```

需要从 constants.ts 导入 `findSectLevel`、`nextSectLevel`、`realmIndex`、`findRealm`。
需要从 repository.ts 导入 `updateSectLevelStatement`。

```typescript
  for (const [resourceId, amount] of Object.entries(next.upgradeCost)) {
```

### 10.6 App.vue 处理 upgrade-sect 事件

在 `App.vue`（或管理 SectScreen 事件的地方）加处理：

```typescript
async function handleUpgradeSect() {
  busy.value = true;
  try {
    const newState = await upgradeSect();
    state.value = newState;
    emit('notify', 'success', '宗门晋升', `宗门已晋升为「${newState.sect.levelName}」！`);
  } catch (e: any) {
    emit('notify', 'warning', '晋升失败', e.message ?? '条件未满足');
  } finally {
    busy.value = false;
  }
}
```
